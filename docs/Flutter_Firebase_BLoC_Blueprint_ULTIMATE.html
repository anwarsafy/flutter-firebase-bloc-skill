import 'package:cloud_firestore/cloud_firestore.dart';
import '../../domain/entities/board_entity.dart';

class BoardModel {
  const BoardModel({
    required this.id, required this.title, required this.color,
    required this.ownerId, required this.memberIds, required this.updatedAt,
    this.createdAt, this.hasPendingWrites = false,
  });

  final String id, title, color, ownerId;
  final List<String> memberIds;
  final DateTime? createdAt;
  final DateTime updatedAt;
  final bool hasPendingWrites;

  factory BoardModel.fromFirestore(DocumentSnapshot<Map<String, dynamic>> doc) {
    final data = doc.data() ?? <String, dynamic>{};
    return BoardModel(
      id: doc.id,
      title: (data['title'] as String?) ?? '',
      color: (data['color'] as String?) ?? '#6366F1',
      ownerId: (data['ownerId'] as String?) ?? '',
      memberIds: List<String>.from((data['memberIds'] as List?) ?? const []),
      createdAt: (data['createdAt'] as Timestamp?)?.toDate(),
      updatedAt: (data['updatedAt'] as Timestamp?)?.toDate() ?? DateTime.now(),
      hasPendingWrites: doc.metadata.hasPendingWrites,
    );
  }

  Map<String, dynamic> toMap() => {
        'title': title, 'color': color, 'ownerId': ownerId,
        'memberIds': memberIds, 'updatedAt': FieldValue.serverTimestamp(),
      };

  BoardEntity toEntity() => BoardEntity(
        id: id, title: title, color: color, ownerId: ownerId,
        memberIds: memberIds, createdAt: createdAt, updatedAt: updatedAt, hasPendingWrites: hasPendingWrites,
      );
}
DART features/boards/data/datasources/boards_remote_datasource.dart 
@LazySingleton(as: BoardsRemoteDatasource)
class BoardsRemoteDatasourceImpl implements BoardsRemoteDatasource {
  const BoardsRemoteDatasourceImpl(this._firestore);
  final FirebaseFirestore _firestore;

  CollectionReference<Map<String, dynamic>> get _raw => _firestore.collection('boards');
  CollectionReference<BoardModel> get _boards => _raw.withConverter<BoardModel>(
        fromFirestore: (snap, _) => BoardModel.fromFirestore(snap),
        toFirestore: (model, _) => model.toMap(),
      );

  @override
  Stream<List<BoardModel>> watchBoards(String userId) => _boards
      .where('memberIds', arrayContains: userId)
      .orderBy('updatedAt', descending: true)
      .limit(50)
      .snapshots(includeMetadataChanges: true)
      .map((s) => s.docs.map((d) => d.data()).toList());

  @override
  Future<BoardModel> createBoard(CreateBoardParams params) async {
    final ref = _raw.doc();
    final model = BoardModel(
      id: ref.id, title: params.title, color: params.color, ownerId: params.userId,
      memberIds: [params.userId], updatedAt: DateTime.now(), hasPendingWrites: true,
    );
    _fireAndLog(ref.set({...model.toMap(), 'createdAt': FieldValue.serverTimestamp()}), 'createBoard');
    return model;
  }

  @override
  Future<void> deleteBoard(String boardId) async {
    final boardRef = _raw.doc(boardId);
    final cards = await boardRef.collection('cards').get();
    final refs = [...cards.docs.map((d) => d.reference), boardRef];
    const chunk = 400;
    for (var i = 0; i < refs.length; i += chunk) {
      final batch = _firestore.batch();
      for (final ref in refs.skip(i).take(chunk)) { batch.delete(ref); }
      _fireAndLog(batch.commit(), 'deleteBoard');
    }
  }

  void _fireAndLog(Future<void> write, String op) {
    unawaited(write.catchError((Object e, StackTrace s) => AppLogger.error('$op rejected', error: e, stack: s)));
  }
}
DART features/boards/data/repositories/boards_repository_impl.dart 
@LazySingleton(as: BoardsRepository)
class BoardsRepositoryImpl implements BoardsRepository {
  const BoardsRepositoryImpl(this._remote);
  final BoardsRemoteDatasource _remote;

  @override
  Stream<Either<AppFailure, List<BoardEntity>>> watchBoards(String userId) async* {
    try {
      yield* _remote.watchBoards(userId).map<Either<AppFailure, List<BoardEntity>>>(
            (models) => Right(models.map((m) => m.toEntity()).toList()),
          );
    } on FirebaseException catch (e) {
      yield Left(e.toFailure());
    }
  }

  @override
  Future<Either<AppFailure, BoardEntity>> createBoard(CreateBoardParams params) async {
    try {
      final model = await _remote.createBoard(params);
      return Right(model.toEntity());
    } on FirebaseException catch (e) {
      return Left(e.toFailure());
    } catch (e, s) {
      AppLogger.error('createBoard failed', error: e, stack: s);
      return Left(UnknownFailure(e.toString()));
    }
  }
}
DART Cursor pagination pattern — startAfterDocument + opaque cursor 
class PaginatedResult<T> {
  const PaginatedResult({required this.items, required this.hasMore, this.cursor});
  final List<T> items;
  final bool hasMore;
  final Object? cursor;
}

Future<PaginatedResult<ActivityModel>> fetchActivityPage({
  required String boardId, Object? cursor, int pageSize = 25,
}) async {
  var query = _firestore
      .collection('boards').doc(boardId).collection('activity')
      .orderBy('createdAt', descending: true).limit(pageSize);
  if (cursor is DocumentSnapshot) query = query.startAfterDocument(cursor);
  final snap = await query.get();
  return PaginatedResult(
    items: snap.docs.map(ActivityModel.fromFirestore).toList(),
    hasMore: snap.docs.length == pageSize,
    cursor: snap.docs.isEmpty ? cursor : snap.docs.last,
  );
}
DART core/storage/storage_service.dart 
@lazySingleton
class StorageService {
  StorageService(this._storage, this._perf);
  final FirebaseStorage _storage;
  final PerfService _perf;

  Future<String> uploadBytes({
    required String path, required Uint8List bytes, required String contentType,
    void Function(double progress)? onProgress,
  }) =>
      _perf.trace('storage_upload', () async {
        final ref = _storage.ref(path);
        final task = ref.putData(bytes, SettableMetadata(contentType: contentType));
        final sub = task.snapshotEvents.listen((s) {
          if (s.totalBytes > 0) onProgress?.call(s.bytesTransferred / s.totalBytes);
        });
        try {
          await task;
          return await ref.getDownloadURL();
        } finally {
          await sub.cancel();
        }
      }, attributes: {'size_bucket': bytes.length < 1048576 ? '<1MB' : '>=1MB'});
}

DART presentation/bloc/boards_bloc.dart 
@injectable
class BoardsBloc extends Bloc<BoardsEvent, BoardsState> {
  BoardsBloc(this._watchBoards, this._createBoard, this._deleteBoard) : super(const BoardsState()) {
    on<WatchBoardsStarted>(_onWatchStarted);
    on<BoardsRefreshed>(_onRefreshed);
    on<CreateBoardRequested>(_onCreateBoard);
    on<DeleteBoardRequested>(_onDeleteBoard);
  }
  final WatchBoards _watchBoards;
  final CreateBoard _createBoard;
  final DeleteBoard _deleteBoard;
  String? _userId;

  Future<void> _onWatchStarted(WatchBoardsStarted e, Emitter<BoardsState> emit) async {
    _userId = e.userId;
    emit(state.copyWith(status: BoardsStatus.loading));
    await emit.forEach(
      _watchBoards(e.userId),
      onData: (result) => result.fold(
        (f) => state.copyWith(status: BoardsStatus.failure, failure: () => f),
        (b) => state.copyWith(status: BoardsStatus.success, boards: b, failure: () => null),
      ),
    );
  }

  Future<void> _onRefreshed(BoardsRefreshed _, Emitter<BoardsState> emit) =>
      _userId == null ? Future.value() : _onWatchStarted(WatchBoardsStarted(_userId!), emit);

  Future<void> _onCreateBoard(CreateBoardRequested e, Emitter<BoardsState> emit) async {
    final res = await _createBoard(CreateBoardParams(title: e.title, color: e.color, userId: e.userId));
    res.fold((f) => emit(state.copyWith(status: BoardsStatus.failure, failure: () => f)), (_) {});
  }

  Future<void> _onDeleteBoard(DeleteBoardRequested e, Emitter<BoardsState> emit) async {
    final res = await _deleteBoard(e.boardId);
    res.fold((f) => emit(state.copyWith(status: BoardsStatus.failure, failure: () => f)), (_) {});
  }
}

DART core/router/app_router.dart 
GoRouter createAppRouter({required AuthBloc authBloc, required FeatureFlagsCubit flags}) {
  return GoRouter(
    initialLocation: RouteNames.splash,
    refreshListenable: Listenable.merge([StreamListenable(authBloc.stream), StreamListenable(flags.stream)]),
    redirect: (context, state) {
      final loc = state.matchedLocation;
      final f = flags.state;
      if (f.maintenanceMode) return loc == RouteNames.maintenance ? null : RouteNames.maintenance;
      if (isBelowMinVersion(f.minSupportedVersion)) {
        return loc == RouteNames.updateRequired ? null : RouteNames.updateRequired;
      }
      if (loc == RouteNames.maintenance || loc == RouteNames.updateRequired) return RouteNames.splash;

      final authState = authBloc.state;
      if (authState.status == AuthStatus.unknown) {
        return loc == RouteNames.splash ? null : RouteNames.splash;
      }
      final isAuth = authState.status == AuthStatus.authenticated;
      final onAuthRoute = loc.startsWith(RouteNames.authPrefix);

      if (!isAuth) return onAuthRoute ? null : RouteNames.welcome;
      if (authState.user?.needsOnboarding ?? false) {
        return loc == RouteNames.onboarding ? null : RouteNames.onboarding;
      }
      if (loc == RouteNames.splash || onAuthRoute || loc == RouteNames.onboarding) {

class AuthRemoteDatasourceImpl implements AuthRemoteDatasource {
  const AuthRemoteDatasourceImpl(this._auth, this._firestore);
  final FirebaseAuth _auth; final FirebaseFirestore _firestore;

  @override
  Stream<UserModel?> watchAuthState() => _auth.authStateChanges().asyncExpand((user) {
        if (user == null) return Stream.value(null);
        return _firestore.collection('users').doc(user.uid).snapshots().map(
              (doc) => doc.exists ? UserModel.fromFirestore(doc) : UserModel.fromFirebaseUser(user),
            );
      });

  @override
  Future<UserModel> signInWithGoogle() async {
    final googleUser = await GoogleSignIn().signIn();
    if (googleUser == null) throw const CancelledException();
    final ga = await googleUser.authentication;
    final cred = await _auth.signInWithCredential(GoogleAuthProvider.credential(accessToken: ga.accessToken, idToken: ga.idToken));
    return _upsertUser(cred.user!);
  }

  @override
  Future<UserModel> signInWithApple() async {
    final rawNonce = generateNonce();
    final sha256Nonce = sha256.convert(utf8.encode(rawNonce)).toString();
    final appleCred = await SignInWithApple.getAppleIDCredential(
      scopes: [AppleIDAuthorizationScopes.email, AppleIDAuthorizationScopes.fullName], nonce: sha256Nonce,
    );
    final cred = await _auth.signInWithCredential(OAuthProvider('apple.com').credential(idToken: appleCred.identityToken, rawNonce: rawNonce));
    final fullName = [appleCred.givenName, appleCred.familyName].whereType<String>().join(' ').trim();
    return _upsertUser(cred.user!, displayName: fullName.isEmpty ? null : fullName);
  }

  @override
  Future<String> sendOtp(String phoneNumber) {
    final completer = Completer<String>();
    unawaited(_auth.verifyPhoneNumber(
      phoneNumber: phoneNumber, timeout: const Duration(seconds: 60),
      verificationCompleted: (_) {},
      verificationFailed: (e) { if (!completer.isCompleted) completer.completeError(e); },
      codeSent: (id, _) { if (!completer.isCompleted) completer.complete(id); },
      codeAutoRetrievalTimeout: (_) {},
    ));
    return completer.future;
  }

  @override
  Future<UserModel> verifyOtp({required String verificationId, required String otp}) async {
    final cred = await _auth.signInWithCredential(PhoneAuthProvider.credential(verificationId: verificationId, smsCode: otp.trim()));
    return _upsertUser(cred.user!);
  }

  @override
  Future<UserModel> signInWithEmail({required String email, required String password}) async {
    final cred = await _auth.signInWithEmailAndPassword(email: email.trim(), password: password);
    return _upsertUser(cred.user!);
  }

  @override
  Future<UserModel> signUpWithEmail({required String email, required String password}) async {
    final cred = await _auth.createUserWithEmailAndPassword(email: email.trim(), password: password);
    return _upsertUser(cred.user!);
  }

  @override
  Future<void> sendPasswordReset(String email) => _auth.sendPasswordResetEmail(email: email.trim());

  @override
  Future<void> deleteAccount() async {
    final user = _auth.currentUser;
    if (user != null) await user.delete();
  }

  Future<UserModel> _upsertUser(User user, {String? displayName}) async {
    final ref = _firestore.collection('users').doc(user.uid);
    await ref.set({
      'id': user.uid, 'name': displayName ?? user.displayName, 'email': user.email,
      'phone': user.phoneNumber, 'photoUrl': user.photoURL, 'lastSignIn': FieldValue.serverTimestamp(),
    }, SetOptions(merge: true));
    return UserModel.fromFirestore(await ref.get());
  }
}

RULES firebase/firestore.rules (Hardened against Privilege Escalation, Scraping & Payload Abuse) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    function isOwner(userId) { return isSignedIn() && request.auth.uid == userId; }
    function isMember(boardData) { return isSignedIn() && request.auth.uid in boardData.memberIds; }
    function unchanged(fields) { return !request.resource.data.diff(resource.data).affectedKeys().hasAny(fields); }
    function onlyChanges(fields) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(fields); }
    function validBoardFields(d) {
      return d.keys().hasOnly(['title', 'color', 'ownerId', 'memberIds', 'createdAt', 'updatedAt'])
        && d.title is string && d.title.size() >= 1 && d.title.size() <= 120
        && d.color is string && d.color.size() <= 16
        && d.memberIds is list && d.memberIds.size() <= 50 && d.ownerId in d.memberIds;
    }

    // Users: prevent self-promotion (`isPro`, `role`, `boardCount` are Admin SDK only!)
    match /users/{userId} {
      allow read: if isOwner(userId);
      allow create: if isOwner(userId)
        && !request.resource.data.keys().hasAny(['isPro', 'role', 'boardCount']);
      allow update: if isOwner(userId)
        && unchanged(['id', 'isPro', 'role', 'boardCount']);
      allow delete: if false; // Cleaned up by Cloud Function on Auth delete
    }

    // Public profiles: allow single-doc lookup (`get`), block full collection dump (`list`)
    match /publicProfiles/{userId} {
      allow get: if isSignedIn();
      allow list: if false;
      allow write: if isOwner(userId) && request.resource.data.keys().hasOnly(['name', 'photoUrl', 'username']);
    }

    match /boards/{boardId} {
      allow read: if isMember(resource.data);
      allow create: if isSignedIn() && request.resource.data.ownerId == request.auth.uid
        && request.resource.data.memberIds == [request.auth.uid] && validBoardFields(request.resource.data);
      allow update: if isMember(resource.data) && unchanged(['ownerId', 'createdAt'])
        && (isOwner(resource.data.ownerId) || unchanged(['memberIds']))
        && validBoardFields(request.resource.data);
      allow delete: if isOwner(resource.data.ownerId);

      match /cards/{cardId} {
        allow read: if isMember(get(/databases/$(database)/documents/boards/$(boardId)).data);
        allow create, update: if isMember(get(/databases/$(database)/documents/boards/$(boardId)).data)
          && request.resource.data.keys().hasOnly(['title', 'description', 'position', 'assigneeId', 'updatedAt', 'createdAt'])
          && request.resource.data.title is string && request.resource.data.title.size() <= 200;
        allow delete: if isMember(get(/databases/$(database)/documents/boards/$(boardId)).data);
      }
    }

    match /notifications/{notifId} {
      allow read: if isOwner(resource.data.userId);
      allow update: if isOwner(resource.data.userId) && onlyChanges(['isRead']);
      allow create, delete: if false;
    }
  }
}

export const deleteBoardRecursive = onCall({ enforceAppCheck: true }, async (req) => {
  const uid = req.auth?.uid;
  const boardId = req.data?.boardId as string | undefined;
  if (!uid) throw new HttpsError('unauthenticated', 'Sign in required');
  if (!boardId) throw new HttpsError('invalid-argument', 'boardId required');
  const ref = admin.firestore().collection('boards').doc(boardId);
  const snap = await ref.get();
  if (!snap.exists || snap.data()?.ownerId !== uid) throw new HttpsError('permission-denied', 'Owner only');
  await admin.firestore().recursiveDelete(ref);
  return { deleted: true };
});

export const onCardAssigned = onDocumentWritten('boards/{boardId}/cards/{cardId}', async (event) => {
  const before = event.data?.before.data();
  const after = event.data?.after.data();
  if (!after || !after.assigneeId || before?.assigneeId === after.assigneeId) return null;
  const assigneeId = after.assigneeId as string;
  const { boardId, cardId } = event.params;

  await admin.firestore().collection('notifications').add({
    userId: assigneeId, title: 'New card assigned to you', body: after.title ?? 'New task',
    data: { type: 'card_assigned', id: cardId, boardId }, isRead: false, createdAt: admin.firestore.FieldValue.serverTimestamp(),
  });

  const userDoc = await admin.firestore().collection('users').doc(assigneeId).get();
  const tokens: string[] = userDoc.data()?.fcmTokens ?? [];
  if (!tokens.length) return null;

  const res = await admin.messaging().sendEachForMulticast({
    tokens, notification: { title: 'New card assigned to you', body: after.title ?? 'New task' },
    data: { type: 'card_assigned', id: cardId, boardId },
    android: { priority: 'high', notification: { channelId: 'high_importance_channel', priority: 'max', sound: 'default' } },
    apns: { payload: { aps: { sound: 'default', badge: 1, contentAvailable: true } } },
  });

  const stale = res.responses
    .map((r, i) => ({ r, t: tokens[i] }))
    .filter(({ r }) => !r.success && ['messaging/registration-token-not-registered', 'messaging/invalid-registration-token'].includes(r.error?.code ?? ''))
    .map(({ t }) => t);
  if (stale.length) await userDoc.ref.update({ fcmTokens: admin.firestore.FieldValue.arrayRemove(...stale) });
  return { sent: res.successCount, removed: stale.length };
});

@lazySingleton
class FcmService {
  FcmService(this._messaging, this._firestore, this._local);
  final FirebaseMessaging _messaging;
  final FirebaseFirestore _firestore;
  final LocalNotificationService _local;

  StreamSubscription<String>? _tokenSub;
  StreamSubscription<RemoteMessage>? _fgSub, _tapSub;
  String? _activeUserId;

  Future<void> initialize(String userId) async {
    if (_activeUserId == userId) return;
    dispose();
    _activeUserId = userId;

    if (!kIsWeb) await _local.init();
    final settings = await _messaging.requestPermission(provisional: true);
    AppLogger.info('FCM permission: ${settings.authorizationStatus}', tag: 'FCM');
    if (settings.authorizationStatus == AuthorizationStatus.denied) {
      await _firestore.collection('users').doc(userId).set({'notificationsEnabled': false}, SetOptions(merge: true));
      return;
    }

    final token = await _messaging.getToken(vapidKey: kIsWeb ? AppConfig.instance.webVapidKey : null);
    if (token != null) await _saveToken(userId, token);

    _tokenSub = _messaging.onTokenRefresh.listen((t) => _saveToken(userId, t));
    _fgSub = FirebaseMessaging.onMessage.listen(_handleForeground);
    _tapSub = FirebaseMessaging.onMessageOpenedApp.listen((m) => NotificationNavigator.navigate(m.data));
  }

  Future<void> _handleForeground(RemoteMessage m) async {
    final n = m.notification;
    if (n == null || kIsWeb) return;
    await _local.show(id: m.hashCode, title: n.title ?? '', body: n.body ?? '', payload: jsonEncode(m.data), highPriority: m.data['priority'] == 'high');
    if (await AppBadgePlus.isSupported()) {
      await AppBadgePlus.updateBadge((await AppBadgePlus.getBadgeCount()) + 1);
    }
  }

  Future<void> _saveToken(String uid, String token) => _firestore.collection('users').doc(uid).set({
        'fcmTokens': FieldValue.arrayUnion([token]), 'notificationsEnabled': true, 'lastTokenUpdatedAt': FieldValue.serverTimestamp(),
      }, SetOptions(merge: true));

  Future<void> removeToken(String uid) async {
    final t = await _messaging.getToken();
    if (t == null) return;
    await _firestore.collection('users').doc(uid).update({'fcmTokens': FieldValue.arrayRemove([t])});
    await _messaging.deleteToken();
  }

  Future<void> clearBadge() async { if (!kIsWeb && await AppBadgePlus.isSupported()) await AppBadgePlus.updateBadge(0); }

  void dispose() {
    unawaited(_tokenSub?.cancel()); unawaited(_fgSub?.cancel()); unawaited(_tapSub?.cancel());
    _tokenSub = _fgSub = _tapSub = null; _activeUserId = null;
  }
}

void _onTap(NotificationResponse response) {
  if (response.payload == null) return;
  NotificationNavigator.navigateFromPayload(response.payload!);
}

abstract final class NotificationNavigator {
  static void navigateFromPayload(String payload) {
    try {
      final decoded = jsonDecode(payload);
      if (decoded is Map<String, dynamic>) navigate(decoded);
    } catch (e, s) {
      AppLogger.error('Invalid notification payload', error: e, stack: s);
    }
  }
  static void navigate(Map<String, dynamic> data) {
    final type = data['type'] as String?, id = data['id'] as String?, boardId = data['boardId'] as String?;
    switch (type) {
      case 'board_invite': if (id != null) AppNavigator.go(RouteNames.boardPath(id));
      case 'card_assigned' || 'comment_added': if (boardId != null) AppNavigator.go(RouteNames.boardPath(boardId));
      default: AppNavigator.go(RouteNames.notifications);
    }
  }
}

@lazySingleton
class PerfService {
  PerfService(this._perf);
  final FirebasePerformance _perf;
  Future<void> init() => _perf.setPerformanceCollectionEnabled(AppConfig.instance.isProd);

  Future<T> trace<T>(String name, Future<T> Function() op, {Map<String, String> attributes = const {}}) async {
    final t = _perf.newTrace(name);
    attributes.forEach(t.putAttribute);
    await t.start();
    try {
      final r = await op(); t.putAttribute('result', 'ok'); return r;
    } catch (_) {
      t.putAttribute('result', 'error'); rethrow;
    } finally {
      await t.stop();
    }
  }
}
